async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):

async def glory(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if len(context.args) == 0:
        await context.bot.send_message(chat_id=update.effective_chat.id, text="Use: /glory 123456789")
        return
    uid = context.args[0]
    await context.bot.send_message(chat_id=update.effective_chat.id, text=f"UID {uid} glory check panren...")
    await asyncio.sleep(1)
    await context.bot.send_message(chat_id=update.effective_chat.id,
    text=f"🔥 UID: {uid}\n👤 Name: Player{uid[-4:]}\n🏆 Glory: 52341\n⭐ Level: 68")

async def song(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if len(context.args) == 0:
        await context.bot.send_message(chat_id=update.effective_chat.id, text="Use: /song Anirudh")
        return
    song_name = " ".join(context.args)
    await context.bot.send_message(chat_id=update.effective_chat.id,
    text=f"🎵 {song_name}\nYouTube: https://youtube.com/results?search_query={song_name}")
